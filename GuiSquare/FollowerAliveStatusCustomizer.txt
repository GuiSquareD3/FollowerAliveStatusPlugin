// NOTE: RBH disables every plugin it does not know about. If you enable this file,
// ALSO add this line to Enable_Plugins in
//   plugins/RosbotHelper/Config/Manager_Config.cs
//   "Turbo.Plugins.GuiSquare.FollowerAliveStatusCustomizer"

using Turbo.Plugins.Default;

namespace Turbo.Plugins.GuiSquare
{
    public class FollowerAliveStatusCustomizer : BasePlugin, ICustomizer
    {
        public FollowerAliveStatusCustomizer()
        {
            Enabled = true;
        }

        public void Customize()
        {
            Hud.RunOnPlugin<FollowerAliveStatusPlugin>(plugin =>
            {
                // ---------------------------------------------------------------
                // 1) Where the icon and counter sit
                // ---------------------------------------------------------------
                // BelowRbhSessionPanel -> below the RBH session panel               [default]
                // UnderPortrait        -> below the hero portrait (top left)
                // LeftOfHealthGlobe    -> left of the health globe (bottom left)
                // RightOfResourceGlobe -> right of the resource globe (bottom right)
                // AboveSkillBar        -> above the skill bar (bottom centre)
                // UnderMinimapClock    -> below the minimap clock (top right)
                // Custom               -> free placement via CustomX / CustomY
                plugin.Position = FollowerStatusIconPosition.BelowRbhSessionPanel;

                // Height of the RBH panel, as a fraction of the minimap's height.
                // THIS is the value to adjust if the icon overlaps the panel or floats
                // too low: increase it to move down, decrease it to move up.
                plugin.RbhPanelHeightRatio = 0.46f;

                // Fine tuning for any position (as a ratio of screen height).
                // plugin.OffsetX = 0.00f;
                // plugin.OffsetY = 0.00f;

                // Fully free placement:
                // plugin.Position = FollowerStatusIconPosition.Custom;
                // plugin.CustomX = 0.50f;
                // plugin.CustomY = 0.90f;

                // ---------------------------------------------------------------
                // 2) Appearance
                // ---------------------------------------------------------------
                // Skull    -> skull tinted with the state colour                    [default]
                // Dot      -> plain filled circle
                // Portrait -> follower's face plus a small state dot
                plugin.IconStyle = FollowerStatusIconStyle.Skull;

                plugin.IconSizeRatio = 0.024f;   // icon size
                plugin.ShowCounter = true;       // show the death counter
                plugin.CounterOnRight = true;    // false = number drawn inside the icon
                plugin.HideCounterWhenZero = false;
                plugin.CounterPrefix = "×"; // reads as "x2" rather than a lone "2"; "" for the bare number
                plugin.BlinkWhenDead = true;
                plugin.HideWhenNoFollower = false; // true = draw nothing when no follower is hired

                // Colours (a, r, g, b, stroke width: 0 = filled)
                // plugin.AliveBrush      = Hud.Render.CreateBrush(230, 40, 200, 70, 0);
                // plugin.DeadBrush       = Hud.Render.CreateBrush(240, 225, 35, 35, 0);
                // plugin.IconDetailBrush = Hud.Render.CreateBrush(255, 10, 10, 12, 0);

                // ---------------------------------------------------------------
                // 3) Counting
                // ---------------------------------------------------------------
                // false = the counter survives quitting to the menu and later games.
                // It only goes back to zero on a TurboHUD restart, or on a reset click.
                plugin.ResetCounterOnNewGame = false;

                // Reset by clicking the icon. Ctrl is required by default because the
                // icon sits in the top-left area where you click to move, and an
                // accidental reset would be unrecoverable. The hover tooltip always
                // states the gesture currently in effect, and the click is swallowed
                // either way so your character never walks under the icon.
                plugin.ResetCounterOnClick = true;
                plugin.ResetClickRequiresCtrl = true;

                // Prefer a plain click with no modifier? Two safe options:
                //   a) drop the modifier entirely
                // plugin.ResetClickRequiresCtrl = false;
                //   b) keep it modifier-free but use a button that cannot be misclicked
                // plugin.ResetClickRequiresCtrl = false;
                // plugin.ResetClickButton = System.Windows.Forms.MouseButtons.Middle;

                plugin.SpeakOnDeath = false;     // spoken alert on every follower death

                // ---------------------------------------------------------------
                // 4) Detection
                // ---------------------------------------------------------------
                // Verified in game: death is seen through IActor.Hitpoints dropping to 0
                // while the actor stays present, so the vanish heuristic is disabled and
                // fast teleport chains cannot produce a false death.
                // plugin.TreatVanishAsDeath = false;

                // Diagnostic panel: off. Set it to true to show the raw signals again.
                plugin.DebugEnabled = false;
            });
        }
    }
}
